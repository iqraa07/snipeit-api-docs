🐍 Python — Export ke Excel
Script lengkap untuk mengambil semua 1.742 assets dan export ke file Excel dengan styling.
📦 Install dulu: pip install requests openpyxl
export_assets.py
import requests, openpyxl, warnings
from datetime import datetime
from openpyxl.styles import Font, PatternFill, Alignment
warnings.filterwarnings('ignore')

BASE  = 'http://10.107.240.9'
TOKEN = 'PASTE_TOKEN_DISINI'
HEADERS = {
    'Authorization': f'Bearer {TOKEN}',
    'Accept': 'application/json'
}

def get_all_assets():
    # Auto-pagination: ambil semua data bertahap
    assets, offset, limit = [], 0, 500
    while True:
        r = requests.get(
            f'{BASE}/api/v1/hardware',
            headers=HEADERS,
            params={'limit': limit, 'offset': offset},
            verify=False
        )
        data  = r.json()
        rows  = data.get('rows', [])
        if not rows: break
        assets.extend(rows)
        total = data.get('total', 0)
        print(f'  Fetched {len(assets)}/{total}...')
        if len(assets) >= total: break
        offset += limit
    return assets

def export_to_excel(assets):
    wb = openpyxl.Workbook()
    ws = wb.active
    ws.title = "Assets"
    headers = [
        'Asset Tag', 'Nama Asset', 'Serial', 'Model',
        'Kategori', 'Status', 'Assigned To', 'Lokasi',
        'Person in Charge', 'Kondisi', 'Order Number',
        'Purchase Date', 'Purchase Cost', 'Notes'
    ]
    ws.append(headers)
    for cell in ws[1]:
        cell.font      = Font(bold=True, color="FFFFFF")
        cell.fill      = PatternFill("solid", fgColor="C0392B")
        cell.alignment = Alignment(horizontal="center")
    for a in assets:
        cf = a.get('custom_fields', {})
        pd = a.get('purchase_date')
        ws.append([
            a.get('asset_tag', ''),
            a.get('name') or '',
            a.get('serial') or '',
            a['model']['name'] if a.get('model') else '',
            a['category']['name'] if a.get('category') else '',
            a['status_label']['name'] if a.get('status_label') else '',
            a['assigned_to']['name'] if a.get('assigned_to') else '',
            a['location']['name'] if a.get('location') else '',
            cf.get('Person in Charge',{}).get('value','') or '',
            cf.get('Condition',{}).get('value','') or '',
            a.get('order_number') or '',
            pd.get('date','') if isinstance(pd,dict) else (pd or ''),
            a.get('purchase_cost') or '',
            a.get('notes') or '',
        ])
    for col in ws.columns:
        w = max(len(str(c.value or '')) for c in col)
        ws.column_dimensions[col[0].column_letter].width = min(w+2,45)
    fname = f'assets_{datetime.now().strftime("%Y%m%d_%H%M")}.xlsx'
    wb.save(fname)
    return fname

assets = get_all_assets()
fname  = export_to_excel(assets)
print(f'✅ Selesai! {len(assets)} assets → {fname}')

💬 WhatsApp API Integration
Kirim rekap atau notifikasi asset ke WhatsApp via Meta Business API.
⚠️
Butuh credentials WA API: PHONE_NUMBER_ID, ACCESS_TOKEN, dan nomor tujuan.
send_wa.py
import requests

PHONE_ID = 'YOUR_PHONE_NUMBER_ID'
WA_TOKEN = 'YOUR_WA_ACCESS_TOKEN'
TO       = '62812xxxxxxx'  # nomor tujuan (format internasional)

def send_wa(message):
    url = f'https://graph.facebook.com/v18.0/{PHONE_ID}/messages'
    r = requests.post(url,
        headers={'Authorization': f'Bearer {WA_TOKEN}'},
        json={
            'messaging_product': 'whatsapp',
            'to': TO,
            'type': 'text',
            'text': {'body': message}
        }
    )
    return r.json()

# Contoh: kirim ringkasan assets
msg = """📊 *Laporan Asset Snipe-IT*
——————————————
Total Assets  : 1.742
Assigned      : 1.205
Ready Deploy  : 537
——————————————
📅 {datetime.now().strftime('%d/%m/%Y %H:%M')}"""

send_wa(msg)
