Export ke Excel — Python

Ambil semua 1.742 assets dengan pagination otomatis dan export ke file Excel berformat.

$
pip install requests openpyxl
export_assets.py 
import requests, openpyxl, warnings
from datetime import datetime
from openpyxl.styles import Font, PatternFill, Alignment
warnings.filterwarnings('ignore')

BASE    = 'http://10.107.240.9'
TOKEN   = 'PASTE_TOKEN_DISINI'
HEADERS = {'Authorization': f'Bearer {TOKEN}', 'Accept': 'application/json'}

def get_all_assets():
    assets, offset, limit = [], 0, 500
    while True:
        r     = requests.get(f'{BASE}/api/v1/hardware', headers=HEADERS,
                             params={'limit': limit, 'offset': offset}, verify=False)
        data  = r.json()
        rows  = data.get('rows', [])
        if not rows: break
        assets.extend(rows)
        total = data.get('total', 0)
        print(f'  {len(assets)}/{total} assets...')
        if len(assets) >= total: break
        offset += limit
    return assets

def export_excel(assets):
    wb = openpyxl.Workbook(); ws = wb.active; ws.title = "Assets"
    cols = ['Asset Tag','Nama','Serial','Model','Kategori','Status',
            'Assigned To','Lokasi','Person in Charge','Kondisi',
            'Order Number','Purchase Date','Cost','Notes']
    ws.append(cols)
    for c in ws[1]:
        c.font = Font(bold=True, color="FFFFFF")
        c.fill = PatternFill("solid", fgColor="C0392B")  # Pertamina red
    for a in assets:
        cf = a.get('custom_fields', {}); pd = a.get('purchase_date')
        ws.append([
            a.get('asset_tag',''), a.get('name') or '', a.get('serial') or '',
            a['model']['name'] if a.get('model') else '',
            a['category']['name'] if a.get('category') else '',
            a['status_label']['name'] if a.get('status_label') else '',
            a['assigned_to']['name'] if a.get('assigned_to') else '',
            a['location']['name'] if a.get('location') else '',
            cf.get('Person in Charge',{}).get('value','') or '',
            cf.get('Condition',{}).get('value','') or '',
            a.get('order_number') or '',
            pd.get('date','') if isinstance(pd,dict) else (pd or ''),
            a.get('purchase_cost') or '', a.get('notes') or '',
        ])
    for col in ws.columns:
        ws.column_dimensions[col[0].column_letter].width = min(
            max(len(str(c.value or '')) for c in col)+2, 45)
    fname = f'assets_{datetime.now().strftime("%Y%m%d_%H%M")}.xlsx'
    wb.save(fname); return fname

assets = get_all_assets()
fname  = export_excel(assets)
print(f'Selesai: {len(assets)} assets -> {fname}')

Integrasi WhatsApp

Kirim rekap atau notifikasi asset ke grup WA via Meta Business API.

Yang dibutuhkan

Konfigurasi WA Business API dari Meta sudah tersedia. Isi tiga variabel berikut di script:

  • PHONE_NUMBER_ID
  • ACCESS_TOKEN
  • TO (nomor tujuan)

Catatan
Nomor tujuan pakai format internasional tanpa + contoh: 628123456789
send_wa.py
import requests
from datetime import datetime

PHONE_ID = 'YOUR_PHONE_NUMBER_ID'
WA_TOKEN = 'YOUR_WA_ACCESS_TOKEN'
TO       = '628123456789'

def send_wa(msg):
    url = f'https://graph.facebook.com/v18.0/{PHONE_ID}/messages'
    return requests.post(url,
        headers={'Authorization': f'Bearer {WA_TOKEN}'},
        json={'messaging_product': 'whatsapp', 'to': TO,
              'type': 'text', 'text': {'body': msg}}
    ).json()

tgl = datetime.now().strftime('%d/%m/%Y %H:%M')
send_wa(f"""Laporan Asset Snipe-IT
Total  : 1.742
Tgl    : {tgl}
Server : http://10.107.240.9""")
